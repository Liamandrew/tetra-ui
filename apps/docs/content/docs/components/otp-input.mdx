## Usage

The API mirrors [shadcn Input OTP](https://ui.shadcn.com/docs/components/base/input-otp) — `InputOTP` maps to `OTPInput`, `InputOTPGroup` to `OTPInputGroup`, and so on.

```tsx
import {
  OTPInput,
  OTPInputGroup,
  OTPInputSeparator,
  OTPInputSlot,
} from "@/components/ui/otp-input";
```

```tsx
<OTPInput maxLength={6} onComplete={(code) => console.log(code)}>
  <OTPInputGroup>
    <OTPInputSlot index={0} />
    <OTPInputSlot index={1} />
    <OTPInputSlot index={2} />
  </OTPInputGroup>
  <OTPInputSeparator />
  <OTPInputGroup>
    <OTPInputSlot index={3} />
    <OTPInputSlot index={4} />
    <OTPInputSlot index={5} />
  </OTPInputGroup>
</OTPInput>
```

## Four digits

A common pattern for PIN codes:

```tsx
<OTPInput maxLength={4} secureTextEntry onComplete={(code) => console.log(code)}>
  <OTPInputSlot index={0} />
  <OTPInputSlot index={1} />
  <OTPInputSlot index={2} />
  <OTPInputSlot index={3} />
</OTPInput>
```

## Pattern

Restrict input to specific character patterns:

```tsx
import { OTPInput, REGEXP_ONLY_DIGITS } from "@/components/ui/otp-input";

<OTPInput maxLength={6} pattern={REGEXP_ONLY_DIGITS}>
  <OTPInputGroup>
    <OTPInputSlot index={0} />
    <OTPInputSlot index={1} />
    <OTPInputSlot index={2} />
  </OTPInputGroup>
  <OTPInputSeparator />
  <OTPInputGroup>
    <OTPInputSlot index={3} />
    <OTPInputSlot index={4} />
    <OTPInputSlot index={5} />
  </OTPInputGroup>
</OTPInput>
```

## Form

Use with `Field` and `FieldControl` for labels and validation:

```tsx
import {
  Field,
  FieldControl,
  FieldErrorMessage,
  FieldLabel,
} from "@/components/ui/form";
import {
  OTPInput,
  OTPInputGroup,
  OTPInputSeparator,
  OTPInputSlot,
} from "@/components/ui/otp-input";

<Field errorMessage={error} invalid={!!error}>
  <FieldLabel>Verification code</FieldLabel>
  <FieldControl>
    <OTPInput maxLength={6} onChange={setValue} value={value}>
      <OTPInputGroup>
        <OTPInputSlot index={0} />
        <OTPInputSlot index={1} />
        <OTPInputSlot index={2} />
      </OTPInputGroup>
      <OTPInputSeparator />
      <OTPInputGroup>
        <OTPInputSlot index={3} />
        <OTPInputSlot index={4} />
        <OTPInputSlot index={5} />
      </OTPInputGroup>
    </OTPInput>
  </FieldControl>
  <FieldErrorMessage />
</Field>
```

### OTPInputGroup

Extends [React Native View](https://reactnative.dev/docs/view#props) props.

### OTPInputSlot

Extends [React Native Pressable](https://reactnative.dev/docs/pressable#props) props. Defaults children to placeholder, value, and caret.

<TypeTable
  type={{
    index: {
      description: "Zero-based slot index (must be less than `maxLength`).",
      type: "number",
      required: true,
    },
  }}
/>

### OTPInputSlotPlaceholder

Extends [`Text`](/docs/components/text) props. Falls back to `placeholder[index]` from `OTPInput`.

### OTPInputSlotValue

Extends [`Text`](/docs/components/text) props. Falls back to the slot character (or `•` when `secureTextEntry`).

### OTPInputSlotCaret

Extends Reanimated `Animated.View` props. Renders only when the slot is active and empty.

### OTPInputSeparator

Extends [React Native View](https://reactnative.dev/docs/view#props) props. Must be rendered under `OTPInput`.
